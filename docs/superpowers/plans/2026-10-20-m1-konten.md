# M1 Konten – Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Benutzerkonten von Ende zu Ende: Registrieren, Login, Logout, Profil ansehen, Gewicht und Wochenziel ändern, Passwort ändern und Konto löschen – im Backend per JWT abgesichert, in der App mit gespeichertem Token.

**Architecture:** Das Backend bekommt die Tabelle `users`, einen `UserService` mit allen Regeln und JWT-Authentifizierung (Ktor `auth-jwt`, BCrypt für Passwörter). Die App speichert den Token in DataStore. Eine `AuthSession` kennt den Anmeldestatus, ein OkHttp-Interceptor hängt den Token an jede Anfrage und meldet bei 401 ab. Die Wurzel-Composable `TouchGrassRoot` zeigt je nach Status Login/Registrierung oder die Tabs.

**Tech Stack:** Backend: Ktor 3.6.0 (`auth`, `auth-jwt`), java-jwt (kommt mit), `at.favre.lib:bcrypt` 0.10.2, Exposed 1.6.0. App: DataStore Preferences 1.2.1, Retrofit 3.0.0, OkHttp 4.12.0, Navigation Compose 2.10.2.

**Spec:** `docs/superpowers/specs/2026-10-20-touchgrass-design.md` · **API-Vertrag:** `docs/api.md`

```kotlin
package de.touchgrass.plugins

import io.ktor.http.HttpStatusCode
import io.ktor.server.application.Application
import io.ktor.server.application.install
import io.ktor.server.plugins.BadRequestException
import io.ktor.server.plugins.statuspages.StatusPages
import io.ktor.server.response.respond
import kotlinx.serialization.Serializable

@Serializable
data class ErrorResponse(val error: String, val message: String, val field: String? = null)

class ApiException(
    val status: HttpStatusCode,
    val error: String,
    override val message: String,
    val field: String? = null,
) : RuntimeException(message)

fun validationError(field: String, message: String): Nothing =
    throw ApiException(HttpStatusCode.BadRequest, "VALIDATION_ERROR", message, field)

fun Application.configureStatusPages() {
    install(StatusPages) {
        exception<ApiException> { call, cause ->
            call.respond(cause.status, ErrorResponse(cause.error, cause.message, cause.field))
        }
        exception<BadRequestException> { call, _ ->
            call.respond(HttpStatusCode.BadRequest, ErrorResponse("VALIDATION_ERROR", "Ungültige Anfrage."))
        }
        exception<Throwable> { call, cause ->
            call.application.environment.log.error("Unerwarteter Fehler", cause)
            call.respond(HttpStatusCode.InternalServerError, ErrorResponse("INTERNAL_ERROR", "Interner Serverfehler"))
        }
        status(HttpStatusCode.NotFound) { call, status ->
            call.respond(status, ErrorResponse("NOT_FOUND", "Nicht gefunden"))
        }
    }
}
```

```kotlin
package de.touchgrass.users

import de.touchgrass.plugins.validationError

object UserValidation {
    private val usernamePattern = Regex("^[A-Za-z0-9_]{3,20}$")
    private val emailPattern = Regex("^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$")

    fun username(value: String) {
        if (!usernamePattern.matches(value)) {
            validationError("username", "Der Benutzername muss 3–20 Zeichen lang sein und darf nur Buchstaben, Ziffern und _ enthalten.")
        }
    }

    fun email(value: String) {
        if (value.length > 254 || !emailPattern.matches(value)) {
            validationError("email", "Bitte gib eine gültige E-Mail-Adresse ein.")
        }
    }

    fun password(value: String, field: String = "password") {
        if (value.length !in 8..64) {
            validationError(field, "Das Passwort muss 8–64 Zeichen lang sein.")
        }
    }

    fun weight(value: Double) {
        if (value !in 30.0..300.0) {
            validationError("weightKg", "Das Gewicht muss zwischen 30 und 300 kg liegen.")
        }
    }

    fun goalTarget(value: Int) {
        if (value <= 0) {
            validationError("goalTarget", "Das Wochenziel muss größer als 0 sein.")
        }
    }
}
```

`backend/src/main/kotlin/de/touchgrass/auth/JwtSecret.kt`:
```kotlin
package de.touchgrass.auth

import java.nio.file.Files
import java.nio.file.Path
import java.security.SecureRandom
import java.util.Base64

object JwtSecret {
    fun load(fromEnvironment: String?, dataDir: Path): String {
        if (!fromEnvironment.isNullOrBlank()) return fromEnvironment
        val file = dataDir.resolve("jwt-secret")
        if (Files.exists(file)) return Files.readString(file).trim()
        val bytes = ByteArray(64).also { SecureRandom().nextBytes(it) }
        val secret = Base64.getEncoder().encodeToString(bytes)
        Files.createDirectories(dataDir)
        Files.writeString(file, secret)
        return secret
    }
}
```

```kotlin
package de.touchgrass.users

import org.jetbrains.exposed.v1.core.ResultRow
import org.jetbrains.exposed.v1.core.dao.id.LongIdTable
import org.jetbrains.exposed.v1.core.eq

import org.jetbrains.exposed.v1.jdbc.selectAll
import org.jetbrains.exposed.v1.jdbc.transactions.transaction
import org.jetbrains.exposed.v1.jdbc.update
import java.math.BigDecimal

object UsersTable : LongIdTable("users") {
    val username = varchar("username", 20)
    val email = varchar("email", 254)
    val passwordHash = varchar("password_hash", 100)
    val weightKg = decimal("weight_kg", 5, 1).nullable()
    val goalType = enumerationByName<GoalType>("goal_type", 20).default(GoalType.MINUTES)
    val goalTarget = integer("goal_target").default(150)
}

data class UserRecord(
    val id: Long,
    val username: String,
    val email: String,
    val passwordHash: String,
    val weightKg: Double?,
    val goalType: GoalType,
    val goalTarget: Int,
)

class UserRepository(private val database: Database) {
    fun findById(id: Long): UserRecord? = transaction(database) {
        UsersTable.selectAll().where { UsersTable.id eq id }.singleOrNull()?.toRecord()
    }

    fun findByLogin(login: String): UserRecord? = transaction(database) {
        val normalized = login.trim().lowercase()
        UsersTable.selectAll()
            .where { (UsersTable.username.lowerCase() eq normalized) or (UsersTable.email eq normalized) }
            .singleOrNull()?.toRecord()
    }

    fun usernameTaken(username: String): Boolean = transaction(database) {
        UsersTable.selectAll().where { UsersTable.username.lowerCase() eq username.lowercase() }.count() > 0
    }

    fun emailTaken(email: String): Boolean = transaction(database) {
        UsersTable.selectAll().where { UsersTable.email eq email }.count() > 0
    }

    fun create(username: String, email: String, passwordHash: String): UserRecord = transaction(database) {
        val id = UsersTable.insertAndGetId {
            it[UsersTable.username] = username
            it[UsersTable.email] = email
            it[UsersTable.passwordHash] = passwordHash
        }.value
        UsersTable.selectAll().where { UsersTable.id eq id }.single().toRecord()
    }

    fun updateProfile(id: Long, weightKg: Double?, goalType: GoalType?, goalTarget: Int?) {
        if (weightKg == null && goalType == null && goalTarget == null) return
        transaction(database) {
            UsersTable.update({ UsersTable.id eq id }) { row ->
                weightKg?.let { row[UsersTable.weightKg] = BigDecimal.valueOf(it) }
                goalType?.let { row[UsersTable.goalType] = it }
                goalTarget?.let { row[UsersTable.goalTarget] = it }
            }
        }
    }

    fun updatePasswordHash(id: Long, passwordHash: String) = transaction(database) {
        UsersTable.update({ UsersTable.id eq id }) { it[UsersTable.passwordHash] = passwordHash }
    }

    fun delete(id: Long) = transaction(database) {
        UsersTable.deleteWhere { UsersTable.id eq id }
    }

    private fun ResultRow.toRecord() = UserRecord(
        id = this[UsersTable.id].value,
        username = this[UsersTable.username],
        email = this[UsersTable.email],
        passwordHash = this[UsersTable.passwordHash],
        weightKg = this[UsersTable.weightKg]?.toDouble(),
        goalType = this[UsersTable.goalType],
        goalTarget = this[UsersTable.goalTarget],
    )
}
```

```kotlin
package de.touchgrass.users

import de.touchgrass.auth.JwtService
import de.touchgrass.auth.PasswordHasher
import de.touchgrass.plugins.ApiException
import io.ktor.http.HttpStatusCode

class UserService(private val users: UserRepository, private val jwtService: JwtService) {
    fun register(request: RegisterRequest): AuthResponse {
        val username = request.username.trim()
        val email = request.email.trim().lowercase()
        UserValidation.username(username)
        UserValidation.email(email)
        UserValidation.password(request.password)
        if (users.usernameTaken(username)) {
            throw ApiException(HttpStatusCode.Conflict, "USERNAME_TAKEN", "Dieser Benutzername ist schon vergeben.", "username")
        }
        if (users.emailTaken(email)) {
            throw ApiException(HttpStatusCode.Conflict, "EMAIL_TAKEN", "Diese E-Mail ist schon registriert.", "email")
        }
        val user = users.create(username, email, PasswordHasher.hash(request.password))
        return AuthResponse(jwtService.createToken(user.id), user.toDto())
    }

    fun login(request: LoginRequest): AuthResponse {
        val user = users.findByLogin(request.login)
        if (user == null || !PasswordHasher.verify(request.password, user.passwordHash)) {
            throw ApiException(HttpStatusCode.Unauthorized, "INVALID_CREDENTIALS", "Benutzername/E-Mail oder Passwort ist falsch.")
        }
        return AuthResponse(jwtService.createToken(user.id), user.toDto())
    }

    private fun UserRecord.toDto() = UserDto(id, username, email, weightKg, goalType, goalTarget)
}
```

```kotlin
package de.touchgrass.users

import de.touchgrass.auth.JwtService
import de.touchgrass.auth.PasswordHasher
import de.touchgrass.plugins.ApiException
import de.touchgrass.plugins.validationError
import io.ktor.http.HttpStatusCode

class UserService(private val users: UserRepository, private val jwtService: JwtService) {
    fun register(request: RegisterRequest): AuthResponse {
        val username = request.username.trim()
        val email = request.email.trim().lowercase()
        UserValidation.username(username)
        UserValidation.email(email)
        UserValidation.password(request.password)
        if (users.usernameTaken(username)) {
            throw ApiException(HttpStatusCode.Conflict, "USERNAME_TAKEN", "Dieser Benutzername ist schon vergeben.", "username")
        }
        if (users.emailTaken(email)) {
            throw ApiException(HttpStatusCode.Conflict, "EMAIL_TAKEN", "Diese E-Mail ist schon registriert.", "email")
        }
        val user = users.create(username, email, PasswordHasher.hash(request.password))
        return AuthResponse(jwtService.createToken(user.id), user.toDto())
    }

    fun login(request: LoginRequest): AuthResponse {
        val user = users.findByLogin(request.login)
        if (user == null || !PasswordHasher.verify(request.password, user.passwordHash)) {
            throw ApiException(HttpStatusCode.Unauthorized, "INVALID_CREDENTIALS", "Benutzername/E-Mail oder Passwort ist falsch.")
        }
        return AuthResponse(jwtService.createToken(user.id), user.toDto())
    }

    fun me(userId: Long): UserDto = requireUser(userId).toDto()

    fun updateMe(userId: Long, request: UpdateMeRequest): UserDto {
        request.weightKg?.let(UserValidation::weight)
        request.goalTarget?.let(UserValidation::goalTarget)
        requireUser(userId)
        users.updateProfile(userId, request.weightKg, request.goalType, request.goalTarget)
        return me(userId)
    }

    fun changePassword(userId: Long, request: ChangePasswordRequest) {
        val user = requireUser(userId)
        if (!PasswordHasher.verify(request.oldPassword, user.passwordHash)) {
            validationError("oldPassword", "Das bisherige Passwort ist falsch.")
        }
        UserValidation.password(request.newPassword, field = "newPassword")
        users.updatePasswordHash(userId, PasswordHasher.hash(request.newPassword))
    }

    fun deleteMe(userId: Long) {
        users.delete(userId)
    }

    private fun requireUser(userId: Long): UserRecord =
        users.findById(userId) ?: throw ApiException(HttpStatusCode.Unauthorized, "UNAUTHORIZED", "Bitte melde dich an.")

    private fun UserRecord.toDto() = UserDto(id, username, email, weightKg, goalType, goalTarget)
}
```

```kotlin
package de.touchgrass.data

import de.touchgrass.data.dto.ErrorDto
import kotlinx.serialization.SerializationException
import retrofit2.HttpException
import java.io.IOException

class ApiException(val code: String, override val message: String, val field: String? = null) : Exception(message)

suspend fun <T> apiCall(block: suspend () -> T): T =
    try {
        block()
    } catch (e: HttpException) {
        throw e.toApiException()
    } catch (e: IOException) {
        throw ApiException("NETWORK", "Server nicht erreichbar")
    } catch (e: SerializationException) {
        throw ApiException("BAD_RESPONSE", "Unerwartete Antwort vom Server")
    }

private fun HttpException.toApiException(): ApiException {
    val error = response()?.errorBody()?.string()
        ?.let { body -> runCatching { apiJson.decodeFromString<ErrorDto>(body) }.getOrNull() }
    return if (error != null) {
        ApiException(error.error, error.message, error.field)
    } else {
        ApiException("HTTP_${code()}", "Serverfehler ${code()}")
    }
}
```

```kotlin
package de.touchgrass.data

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.stringPreferencesKey
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

val Context.authDataStore: DataStore<Preferences> by preferencesDataStore(name = "auth")

interface TokenStore {
    val token: Flow<String?>
    suspend fun save(token: String)
    suspend fun clear()
}

class DataStoreTokenStore(private val dataStore: DataStore<Preferences>) : TokenStore {
    override val token: Flow<String?> = dataStore.data.map { it[TOKEN] }

    override suspend fun save(token: String) {
        dataStore.edit { it[TOKEN] = token }
    }

    override suspend fun clear() {
        dataStore.edit { it.remove(TOKEN) }
    }

    private companion object {
        val TOKEN = stringPreferencesKey("token")
    }
}
```

```kotlin
package de.touchgrass.data

import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.map
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

sealed interface AuthState {
    data object Loading : AuthState
    data object LoggedOut : AuthState
    data class LoggedIn(val token: String) : AuthState
}

class AuthSession(private val tokenStore: TokenStore, private val scope: CoroutineScope) {
    val state: StateFlow<AuthState> = tokenStore.token
        .map { token -> if (token == null) AuthState.LoggedOut else AuthState.LoggedIn(token) }
        .stateIn(scope, SharingStarted.Eagerly, AuthState.Loading)

    fun currentToken(): String? = (state.value as? AuthState.LoggedIn)?.token

    suspend fun logIn(token: String) = tokenStore.save(token)

    suspend fun logOut() = tokenStore.clear()

    fun onUnauthorized() {
        scope.launch { tokenStore.clear() }
    }
}
```

```kotlin
package de.touchgrass.data

import okhttp3.Interceptor
import okhttp3.Response

class AuthInterceptor(private val session: AuthSession) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val token = session.currentToken()
        val request = chain.request().newBuilder()
            .apply { if (token != null) header("Authorization", "Bearer $token") }
            .build()
        val response = chain.proceed(request)
        if (response.code == 401 && token != null) session.onUnauthorized()
        return response
    }
}
```

```kotlin
package de.touchgrass.ui.auth

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import de.touchgrass.data.ApiException
import de.touchgrass.data.AuthSession
import de.touchgrass.data.SportApi
import de.touchgrass.data.apiCall
import de.touchgrass.data.dto.RegisterRequest
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class RegisterUiState(
    val username: String = "",
    val email: String = "",
    val password: String = "",
    val loading: Boolean = false,
    val fieldErrors: Map<String, String> = emptyMap(),
    val error: String? = null,
)

class RegisterViewModel(private val api: SportApi, private val session: AuthSession) : ViewModel() {
    private val _state = MutableStateFlow(RegisterUiState())
    val state: StateFlow<RegisterUiState> = _state.asStateFlow()

    fun onUsernameChange(value: String) = edit("username") { copy(username = value) }

    fun onEmailChange(value: String) = edit("email") { copy(email = value) }

    fun onPasswordChange(value: String) = edit("password") { copy(password = value) }

    fun submit() {
        val current = _state.value
        if (current.loading) return
        _state.update { it.copy(loading = true, fieldErrors = emptyMap(), error = null) }
        viewModelScope.launch {
            try {
                val request = RegisterRequest(current.username.trim(), current.email.trim(), current.password)
                session.logIn(apiCall { api.register(request) }.token)
            } catch (e: ApiException) {
                _state.update {
                    if (e.field != null) it.copy(loading = false, fieldErrors = mapOf(e.field to e.message))
                    else it.copy(loading = false, error = e.message)
                }
            }
        }
    }

    private fun edit(field: String, change: RegisterUiState.() -> RegisterUiState) =
        _state.update { it.change().copy(fieldErrors = it.fieldErrors - field, error = null) }
}
```

```kotlin
package de.touchgrass.ui.profile

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import de.touchgrass.data.ApiException
import de.touchgrass.data.AuthSession
import de.touchgrass.data.SportApi
import de.touchgrass.data.apiCall
import de.touchgrass.data.dto.ChangePasswordRequest
import de.touchgrass.data.dto.GoalType
import de.touchgrass.data.dto.UpdateMeRequest
import de.touchgrass.data.dto.UserDto
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class SettingsUiState(
    val loading: Boolean = true,
    val weight: String = "",
    val goalType: GoalType = GoalType.MINUTES,
    val goalTarget: String = "",
    val oldPassword: String = "",
    val newPassword: String = "",
    val fieldErrors: Map<String, String> = emptyMap(),
    val message: String? = null,
    val error: String? = null,
)

class SettingsViewModel(private val api: SportApi, private val session: AuthSession) : ViewModel() {
    private val _state = MutableStateFlow(SettingsUiState())
    val state: StateFlow<SettingsUiState> = _state.asStateFlow()

    init {
        viewModelScope.launch {
            try {
                showProfile(apiCall { api.me() })
            } catch (e: ApiException) {
                _state.update { it.copy(loading = false, error = e.message) }
            }
        }
    }

    fun onWeightChange(value: String) = edit("weightKg") { copy(weight = value) }

    fun onGoalTypeChange(value: GoalType) = edit("goalType") { copy(goalType = value) }

    fun onGoalTargetChange(value: String) = edit("goalTarget") { copy(goalTarget = value) }

    fun onOldPasswordChange(value: String) = edit("oldPassword") { copy(oldPassword = value) }

    fun onNewPasswordChange(value: String) = edit("newPassword") { copy(newPassword = value) }

    fun saveProfile() {
        val current = _state.value
        val weightText = current.weight.trim()
        val weight = if (weightText.isEmpty()) null else weightText.replace(',', '.').toDoubleOrNull()
            ?: return showFieldError("weightKg", "Bitte eine Zahl eingeben, z. B. 78,5.")
        val target = current.goalTarget.trim().toIntOrNull()
            ?: return showFieldError("goalTarget", "Bitte eine ganze Zahl eingeben.")
        viewModelScope.launch {
            try {
                showProfile(apiCall { api.updateMe(UpdateMeRequest(weight, current.goalType, target)) })
                _state.update { it.copy(message = "Gespeichert.") }
            } catch (e: ApiException) {
                showError(e)
            }
        }
    }

    fun changePassword() {
        val current = _state.value
        viewModelScope.launch {
            try {
                apiCall { api.changePassword(ChangePasswordRequest(current.oldPassword, current.newPassword)) }
                _state.update { it.copy(oldPassword = "", newPassword = "", message = "Passwort geändert.") }
            } catch (e: ApiException) {
                showError(e)
            }
        }
    }

    fun logOut() {
        viewModelScope.launch { session.logOut() }
    }

    fun deleteAccount() {
        viewModelScope.launch {
            try {
                apiCall { api.deleteMe() }
                session.logOut()
            } catch (e: ApiException) {
                showError(e)
            }
        }
    }

    private fun showProfile(user: UserDto) = _state.update {
        it.copy(
            loading = false,
            weight = user.weightKg?.let(::formatWeight) ?: "",
            goalType = user.goalType,
            goalTarget = user.goalTarget.toString(),
        )
    }

    private fun edit(field: String, change: SettingsUiState.() -> SettingsUiState) =
        _state.update { it.change().copy(fieldErrors = it.fieldErrors - field, message = null, error = null) }

    private fun showFieldError(field: String, message: String) =
        _state.update { it.copy(fieldErrors = it.fieldErrors + (field to message), message = null) }

    private fun showError(e: ApiException) {
        if (e.field != null) showFieldError(e.field, e.message)
        else _state.update { it.copy(error = e.message, message = null) }
    }
}
```

```kotlin
    Tab(MapRoute, "🗺️", "Karte"),
    Tab(RecordRoute, "➕", "Aktivität"),
    Tab(AchievementsRoute, "🏆", "Erfolge"),
    Tab(ProfileGraph, "👤", "Profil"),
)

@Composable
fun TouchGrassNavigation(container: AppContainer) {
    val navController = rememberNavController()
    val backStackEntry by navController.currentBackStackEntryAsState()
    val currentDestination = backStackEntry?.destination

    Scaffold(
        bottomBar = {
            NavigationBar {
                tabs.forEach { tab ->
                    NavigationBarItem(
                        selected = currentDestination?.hierarchy?.any { it.hasRoute(tab.route::class) } == true,
                        onClick = {
                            navController.navigate(tab.route) {
                                popUpTo(navController.graph.findStartDestination().id) { saveState = true }
                                launchSingleTop = true
                                restoreState = true
                            }
                        },
                        icon = { Text(tab.emoji) },
                        label = { Text(tab.label) },
                    )
                }
            }
        },
    ) { innerPadding ->
        NavHost(
            navController = navController,
            startDestination = MapRoute,
            modifier = Modifier.padding(innerPadding),
        ) {
            composable<MapRoute> {
                PlaceholderScreen("Karte", "Die Karte kommt in M3.") {
                    ServerStatusCard(container.api)
                }
            }
            composable<RecordRoute> { PlaceholderScreen("Aktivität", "Formular kommt in M2, Timer in M3.") }
            composable<AchievementsRoute> { PlaceholderScreen("Erfolge", "Achievements, Wochenziel und Leaderboard kommen in M4.") }
            navigation<ProfileGraph>(startDestination = ProfileRoute) {
                composable<ProfileRoute> {
                    ProfileScreen(
                        viewModel = viewModel { ProfileViewModel(container.api) },
                        onOpenSettings = { navController.navigate(SettingsRoute) },
                    )
                }
                composable<SettingsRoute> {
                    SettingsScreen(
                        viewModel = viewModel { SettingsViewModel(container.api, container.session) },
                        onBack = { navController.popBackStack() },
                    )
                }
            }
        }
    }
}
```
