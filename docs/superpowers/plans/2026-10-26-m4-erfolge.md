**Goal:** Achievements freischalten, Wochenziel und Serie anzeigen und ein Leaderboard nach Woche/Monat und Punkten/Minuten bieten. In der App kommt das in den 🏆-Tab, neue Achievements erscheinen nach dem Speichern als Dialog.

**Architecture:** Das Backend bekommt reine Rechenregeln in `logic/` (Serie, Achievement-Regeln, Ranglisten-Plätze, Monatsgrenzen), die Tabellen `achievements` und `user_achievements` (Migration V4 mit den sechs Start-Achievements) und die Endpunkte `GET /achievements`, `/progress` und `/leaderboard`. `POST /activities` liefert jetzt `newAchievements`. `Application.module` bekommt eine `Clock`, damit die Tests Woche und Monat festlegen können. Die App bekommt im 🏆-Tab drei Reiter (Übersicht, Achievements, Leaderboard) mit einem gemeinsamen ViewModel.

**Tech Stack:** Backend: Exposed 1.6.0 (`sum()`, `groupBy`, `insertIgnore`), Flyway, Ktor 3.6.0. App: Material3 `PrimaryTabRow`, `LazyVerticalGrid`, `LinearProgressIndicator`, `FilterChip`, Retrofit-`@Query` mit Enums.

## Global Constraints

- Alle Commits direkt auf `master`, keine Branches, nicht pushen. Commit-Nachrichten auf Deutsch mit Trailer `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Alle Tages-, Wochen- und Monatsgrenzen in `Europe/Berlin`. Woche = Montag 00:00 bis Montag 00:00, Monat = Erster 00:00 bis nächster Erster 00:00.
- Achievement-Regeln (aus der Spec): `TOTAL_ACTIVITIES`, `TOTAL_MINUTES`, `TOTAL_KCAL`, `SPORT_ACTIVITIES` (mit `sport_code`), `STREAK_DAYS` (= längste Serie). Erfüllt bei `Wert ≥ threshold`. Freigeschaltet bleibt freigeschaltet, auch wenn Aktivitäten gelöscht werden.

- Leaderboard: Zeitraum `WEEK` | `MONTH` (Standard `WEEK`), Wertung `POINTS` | `MINUTES` (Standard `POINTS`), kcal wird nicht gewertet. Alle mit mindestens einer Aktivität im Zeitraum, Top 50 plus eigener Platz, gleicher Wert gleicher Rang (1, 2, 2, 4). Private Aktivitäten zählen mit, sichtbar sind nur Benutzername und Wert.

    fun date(moment: OffsetDateTime): LocalDate = moment.atZoneSameInstant(ZONE).toLocalDate()

    fun weekOf(moment: OffsetDateTime): TimeRange {
        val monday = date(moment).with(TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY))
        return TimeRange(startOfDay(monday), startOfDay(monday.plusWeeks(1)))
    }

    fun monthOf(moment: OffsetDateTime): TimeRange {
        val first = date(moment).withDayOfMonth(1)
        return TimeRange(startOfDay(first), startOfDay(first.plusMonths(1)))

package de.touchgrass.logic

import java.time.LocalDate

object Streak {
    fun current(days: Set<LocalDate>, today: LocalDate): Int {
        var day = if (today in days) today else today.minusDays(1)
        var count = 0
        while (day in days) {
            count++
            day = day.minusDays(1)
        }
        return count
    }

    fun longest(days: Set<LocalDate>): Int = days
        .filter { it.minusDays(1) !in days }
        .maxOfOrNull { start -> generateSequence(start) { it.plusDays(1) }.takeWhile { it in days }.count() }
        ?: 0

    fun weekDays(days: Set<LocalDate>, monday: LocalDate): List<Boolean> = (0L..6L).map { monday.plusDays(it) in days }
}
```

package de.touchgrass.logic

enum class Requirement { TOTAL_ACTIVITIES, TOTAL_MINUTES, TOTAL_KCAL, SPORT_ACTIVITIES, STREAK_DAYS }

data class UserStats(
    val activities: Int = 0,
    val minutes: Int = 0,
    val kcal: Int = 0,
    val activitiesPerSport: Map<String, Int> = emptyMap(),
    val longestStreak: Int = 0,
) {
    fun valueFor(requirement: Requirement, sportCode: String?): Int = when (requirement) {
        Requirement.TOTAL_ACTIVITIES -> activities
        Requirement.TOTAL_MINUTES -> minutes
        Requirement.TOTAL_KCAL -> kcal
        Requirement.SPORT_ACTIVITIES -> activitiesPerSport[sportCode] ?: 0
        Requirement.STREAK_DAYS -> longestStreak
    }
}

object AchievementRules {
    fun isReached(value: Int, threshold: Int): Boolean = value >= threshold

    // Freigeschaltet bleibt freigeschaltet, auch wenn später Aktivitäten gelöscht werden.
    fun progress(value: Int, threshold: Int, unlocked: Boolean): Int = if (unlocked) threshold else minOf(value, threshold)
}
```

package de.touchgrass.logic

data class Score(val userId: Long, val username: String, val value: Int)

data class RankedScore(val rank: Int, val score: Score)

object Leaderboard {
    const val TOP = 50

    fun rank(scores: List<Score>): List<RankedScore> {
        val sorted = scores.sortedWith(compareByDescending<Score> { it.value }.thenBy { it.username.lowercase() })
        var rank = 0
        return sorted.mapIndexed { index, score ->
            if (index == 0 || sorted[index - 1].value != score.value) rank = index + 1
            RankedScore(rank, score)
        }
    }
}

import java.time.OffsetDateTime

object AchievementsTable : Table("achievements") {
    val code = varchar("code", 30)
    val name = varchar("name", 50)
    val description = varchar("description", 200)
    val emoji = varchar("emoji", 10)
    val requirement = enumerationByName<Requirement>("requirement", 30)
    val threshold = integer("threshold")
    val sportCode = varchar("sport_code", 30).nullable()
    override val primaryKey = PrimaryKey(code)
}

object UserAchievementsTable : Table("user_achievements") {
    val userId = long("user_id")
    val achievementCode = varchar("achievement_code", 30)
    val unlockedAt = timestampWithTimeZone("unlocked_at")
    override val primaryKey = PrimaryKey(userId, achievementCode)
}

data class AchievementRecord(
    val code: String,
    val name: String,
    val description: String,
    val emoji: String,
    val requirement: Requirement,
    val threshold: Int,
    val sportCode: String?,
)

class AchievementRepository(private val database: Database) {
    fun all(): List<AchievementRecord> = transaction(database) {
        AchievementsTable.selectAll()
            .orderBy(AchievementsTable.threshold to SortOrder.ASC, AchievementsTable.code to SortOrder.ASC)
            .map { it.toRecord() }
    }

    fun unlockedAt(userId: Long): Map<String, OffsetDateTime> = transaction(database) {
        UserAchievementsTable.selectAll()
            .where { UserAchievementsTable.userId eq userId }
            .associate { it[UserAchievementsTable.achievementCode] to it[UserAchievementsTable.unlockedAt] }
    }

    fun unlock(userId: Long, codes: List<String>, at: OffsetDateTime) {
        if (codes.isEmpty()) return
        transaction(database) {
            codes.forEach { code ->
                UserAchievementsTable.insertIgnore {
                    it[UserAchievementsTable.userId] = userId
                    it[achievementCode] = code
                    it[unlockedAt] = at
                }
            }
        }
    }

    private fun ResultRow.toRecord() = AchievementRecord(
        code = this[AchievementsTable.code],
        name = this[AchievementsTable.name],
        description = this[AchievementsTable.description],
        emoji = this[AchievementsTable.emoji],
        requirement = this[AchievementsTable.requirement],
        threshold = this[AchievementsTable.threshold],
        sportCode = this[AchievementsTable.sportCode],
    )
}

package de.touchgrass.achievements

import de.touchgrass.activities.ActivityRepository
import de.touchgrass.logic.AchievementRules
import de.touchgrass.logic.BerlinTime
import de.touchgrass.logic.Streak
import de.touchgrass.logic.UserStats
import java.time.Clock
import java.time.OffsetDateTime
import java.time.temporal.ChronoUnit

class AchievementService(
    private val achievements: AchievementRepository,
    private val activities: ActivityRepository,
    private val clock: Clock = Clock.systemUTC(),
) {
    // Auch die Liste schaltet frei, sonst zählten Aktivitäten von vor der Einführung erst mit der nächsten neuen.
    fun list(userId: Long): List<AchievementDto> = evaluate(userId).all

    fun unlockNew(userId: Long): List<AchievementDto> = evaluate(userId).newlyUnlocked

    private class Evaluation(val all: List<AchievementDto>, val newlyUnlocked: List<AchievementDto>)

    private fun evaluate(userId: Long): Evaluation {
        val stats = statsFor(userId)
        val all = achievements.all()
        val unlocked = achievements.unlockedAt(userId)
        val now = OffsetDateTime.now(clock).truncatedTo(ChronoUnit.SECONDS)
        val fresh = all.filter { it.code !in unlocked && AchievementRules.isReached(stats.valueFor(it.requirement, it.sportCode), it.threshold) }
        achievements.unlock(userId, fresh.map { it.code }, now)

        val unlockedNow = unlocked + fresh.associate { it.code to now }
        val dtos = all.map { it.toDto(stats, unlockedNow[it.code]) }
        val freshCodes = fresh.map { it.code }.toSet()
        return Evaluation(dtos, dtos.filter { it.code in freshCodes })
    }

    private fun statsFor(userId: Long): UserStats {
        val all = activities.list(userId)
        return UserStats(
            activities = all.size,
            minutes = all.sumOf { it.durationMin },
            kcal = all.sumOf { it.caloriesKcal },
            activitiesPerSport = all.groupingBy { it.sportCode }.eachCount(),
            longestStreak = Streak.longest(all.map { BerlinTime.date(it.startedAt) }.toSet()),
        )
    }

    private fun AchievementRecord.toDto(stats: UserStats, unlockedAt: OffsetDateTime?) = AchievementDto(
        code = code,
        name = name,
        description = description,
        emoji = emoji,
        threshold = threshold,
        progress = AchievementRules.progress(stats.valueFor(requirement, sportCode), threshold, unlocked = unlockedAt != null),
        unlockedAt = unlockedAt?.let(BerlinTime::format),
    )
}

class ProgressService(
    private val users: UserRepository,
    private val activities: ActivityRepository,
    private val clock: Clock = Clock.systemUTC(),

    fun progress(userId: Long): ProgressDto {
        val user = users.findById(userId) ?: throw ApiException(HttpStatusCode.Unauthorized, "UNAUTHORIZED", "Bitte melde dich an.")
        val now = OffsetDateTime.now(clock)
        val week = BerlinTime.weekOf(now)
        val thisWeek = activities.inRange(userId, week)
        val totals = WeekTotals(thisWeek.sumOf { it.durationMin }, thisWeek.size, thisWeek.sumOf { it.caloriesKcal })

package de.touchgrass.progress

import de.touchgrass.activities.ActivitiesTable
import de.touchgrass.logic.Score
import de.touchgrass.logic.TimeRange
import de.touchgrass.users.UsersTable
import org.jetbrains.exposed.v1.core.JoinType
import org.jetbrains.exposed.v1.core.and
import org.jetbrains.exposed.v1.core.greaterEq
import org.jetbrains.exposed.v1.core.less
import org.jetbrains.exposed.v1.core.sum
import org.jetbrains.exposed.v1.jdbc.Database
import org.jetbrains.exposed.v1.jdbc.select
import org.jetbrains.exposed.v1.jdbc.transactions.transaction

class LeaderboardRepository(private val database: Database) {
    fun scores(range: TimeRange, metric: LeaderboardMetric): List<Score> = transaction(database) {
        val total = when (metric) {
            LeaderboardMetric.POINTS -> ActivitiesTable.points.sum()
            LeaderboardMetric.MINUTES -> ActivitiesTable.durationMin.sum()
        }
        ActivitiesTable.join(UsersTable, JoinType.INNER, ActivitiesTable.userId, UsersTable.id)
            .select(UsersTable.id, UsersTable.username, total)
            .where { (ActivitiesTable.startedAt greaterEq range.from) and (ActivitiesTable.startedAt less range.until) }
            .groupBy(UsersTable.id, UsersTable.username)
            .map { Score(it[UsersTable.id].value, it[UsersTable.username], it[total] ?: 0) }
    }
}

package de.touchgrass.progress

import de.touchgrass.logic.BerlinTime
import de.touchgrass.logic.Leaderboard
import de.touchgrass.logic.RankedScore
import de.touchgrass.plugins.validationError
import java.time.Clock
import java.time.OffsetDateTime

class LeaderboardService(
    private val leaderboard: LeaderboardRepository,
    private val clock: Clock = Clock.systemUTC(),
) {
    fun leaderboard(userId: Long, period: String?, metric: String?): LeaderboardDto {
        val chosenPeriod = choice(period, LeaderboardPeriod.WEEK, "period", "Zeitraum bitte als WEEK oder MONTH angeben.")
        val chosenMetric = choice(metric, LeaderboardMetric.POINTS, "metric", "Wertung bitte als POINTS oder MINUTES angeben.")
        val now = OffsetDateTime.now(clock)
        val range = when (chosenPeriod) {
            LeaderboardPeriod.WEEK -> BerlinTime.weekOf(now)
            LeaderboardPeriod.MONTH -> BerlinTime.monthOf(now)
        }
        val ranked = Leaderboard.rank(leaderboard.scores(range, chosenMetric))
        return LeaderboardDto(
            period = chosenPeriod,
            metric = chosenMetric,
            entries = ranked.take(Leaderboard.TOP).map { it.toDto() },
            me = ranked.firstOrNull { it.score.userId == userId }?.toDto(),
        )
    }

    private inline fun <reified T : Enum<T>> choice(value: String?, default: T, field: String, message: String): T =
        if (value == null) default else enumValues<T>().firstOrNull { it.name == value } ?: validationError(field, message)

    private fun RankedScore.toDto() = LeaderboardEntryDto(rank, score.username, score.value)
}

package de.touchgrass.ui.achievements

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import de.touchgrass.data.ApiException
import de.touchgrass.data.SportApi
import de.touchgrass.data.apiCall
import de.touchgrass.data.dto.AchievementDto

import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class AchievementsUiState(
    val progress: UiState<ProgressDto> = UiState.Loading,
    val achievements: UiState<List<AchievementDto>> = UiState.Loading,
    val leaderboard: UiState<LeaderboardDto> = UiState.Loading,
    val period: LeaderboardPeriod = LeaderboardPeriod.WEEK,
    val metric: LeaderboardMetric = LeaderboardMetric.POINTS,
)

class AchievementsViewModel(private val api: SportApi) : ViewModel() {
    private val _state = MutableStateFlow(AchievementsUiState())
    val state: StateFlow<AchievementsUiState> = _state.asStateFlow()

    fun refresh() {
        viewModelScope.launch {
            val progress = load { api.progress() }
            _state.update { it.copy(progress = progress) }
        }
        viewModelScope.launch {
            val achievements = load { api.achievements() }
            _state.update { it.copy(achievements = achievements) }
        }
        loadLeaderboard()
    }

    fun onPeriodChange(period: LeaderboardPeriod) {
        _state.update { it.copy(period = period) }
        loadLeaderboard()
    }

    fun onMetricChange(metric: LeaderboardMetric) {
        _state.update { it.copy(metric = metric) }
        loadLeaderboard()
    }

    private fun loadLeaderboard() {
        val period = _state.value.period
        val metric = _state.value.metric
        _state.update { it.copy(leaderboard = UiState.Loading) }
        viewModelScope.launch {
            val leaderboard = load { api.leaderboard(period, metric) }
            // Eine ältere Antwort darf die Auswahl nicht überschreiben, wenn inzwischen umgeschaltet wurde.
            _state.update { if (it.period == period && it.metric == metric) it.copy(leaderboard = leaderboard) else it }
        }
    }

    private suspend fun <T> load(call: suspend () -> T): UiState<T> =
        try {
            UiState.Content(apiCall(call))
        } catch (e: ApiException) {
            UiState.Error(e.message)
        }
}

fun WeeklyGoalDto.fraction(): Float = if (target <= 0) 0f else (current.toFloat() / target).coerceIn(0f, 1f)

fun WeeklyGoalDto.label(): String =
    "$current / $target ${type.label()}" + if (current >= target) " – geschafft! 🎉" else ""

fun daysLabel(days: Int): String = if (days == 1) "1 Tag" else "$days Tage"

fun AchievementDto.progressLabel(): String =
    unlockedAt?.let { "Freigeschaltet am ${formatDate(OffsetDateTime.parse(it).toLocalDate())}" } ?: "$progress / $threshold"

fun LeaderboardEntryDto.valueLabel(metric: LeaderboardMetric): String = when (metric) {
    LeaderboardMetric.POINTS -> "$value Punkte"
    LeaderboardMetric.MINUTES -> "$value min"
}

fun LeaderboardDto.ownEntryBelowList(): LeaderboardEntryDto? = me?.takeIf { own -> entries.none { it.username == own.username } }

        }

            Text(state.message, color = MaterialTheme.colorScheme.error)
            TextButton(onClick = onRetry) { Text("Erneut versuchen") }

            )

                onClick = { onPeriodChange(LeaderboardPeriod.MONTH) },
                label = { Text("Monat") },
            )
        }
        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            FilterChip(
                selected = state.metric == LeaderboardMetric.POINTS,
                onClick = { onMetricChange(LeaderboardMetric.POINTS) },
                label = { Text("Punkte") },
            )
            FilterChip(
                selected = state.metric == LeaderboardMetric.MINUTES,
                onClick = { onMetricChange(LeaderboardMetric.MINUTES) },
                label = { Text("Minuten") },
            )
        }
        Section(state.leaderboard, onRetry) { LeaderboardList(it) }
    }
}

@Composable
private fun LeaderboardList(board: LeaderboardDto) {
    if (board.entries.isEmpty()) {
        Text("In diesem Zeitraum hat noch niemand Sport eingetragen.")
        return
    }
    val ownName = board.me?.username
    val ownBelow = board.ownEntryBelowList()
    LazyColumn {
        items(board.entries, key = { it.username }) { entry ->
            LeaderboardRow(entry, board.metric, highlighted = entry.username == ownName)
        }
        if (ownBelow != null) {
            item { HorizontalDivider() }
            item { LeaderboardRow(ownBelow, board.metric, highlighted = true) }
        }
    }
}

@Composable
private fun LeaderboardRow(entry: LeaderboardEntryDto, metric: LeaderboardMetric, highlighted: Boolean) {
    ListItem(
        leadingContent = { Text("${entry.rank}.", style = MaterialTheme.typography.titleMedium) },
        headlineContent = { Text(entry.username) },
        trailingContent = { Text(entry.valueLabel(metric)) },
        colors = if (highlighted) {
            ListItemDefaults.colors(containerColor = MaterialTheme.colorScheme.primaryContainer)
        } else {
            ListItemDefaults.colors()
        },
    )
}
